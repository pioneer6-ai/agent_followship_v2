# Patient Follow-up Agent - Python Dependencies
#
# ONE command installs everything this project needs:
#
#     pip install -r requirements.txt
#
# Nothing else has to be installed afterwards -- the dashboard, the offline
# demo, the patient-list upload, the AWS messaging tools and the live-Claude
# tool-use loop all work from this list alone. The packages that are optional
# *at runtime* (anthropic, boto3/botocore, openpyxl, certifi) are real pins
# rather than comments, because "optional" there means "the agent degrades
# gracefully without it", not "you may have to install it later".
#
# Requires Python 3.10 or newer. The floor is set by anthropic==1.8.0, which
# declares ``Requires-Python >=3.10``; every other pin in this file still
# accepts 3.9. On 3.9 pip cannot resolve the list at all.
#
# Deliberately NOT pip-installable -- and not needed to run anything:
#   pandoc + Google Chrome, used only by proposal/scripts/build_pdf.py to
#   regenerate the proposal/evidence PDFs. See the README "Dependencies".

# Core Framework
flask==3.0.0
werkzeug==3.0.1

# Date/Time handling. Not imported by this project's own code -- it is an
# explicit pin of botocore's requirement (botocore needs python-dateutil>=2.1),
# kept visible so the resolved version is reproducible.
python-dateutil==2.8.2

# IANA timezone database for Windows. ZoneInfo (scheduling/calendar_service.py)
# needs this on Windows; Linux/macOS ship the tzdata natively.
tzdata==2024.1

# For LLM integration — used by utils/llm_parser.py for intelligent patient
# data parsing (handles messy/swapped field values). Also drives the openai-
# compatible path in tools/llm_providers.py (Gemini, vLLM, LiteLLM, etc.).
openai==1.51.0
# openai 1.51.0 passes the removed ``proxies=`` argument to httpx, so httpx 0.28
# (which dropped it) breaks every request with "Client.__init__() got an
# unexpected keyword argument 'proxies'". The failure is silent from the
# caller's side -- the parser logs it and falls back to the rules -- so the
# working resolver is pinned explicitly rather than left to chance.
# httpx is an openai requirement (openai: httpx>=0.23,<1) and nothing else in
# this project uses it; anthropic/mcp use the separate httpx2 package.
httpx==0.27.2
# Required for LIVE Claude (or any Anthropic-compatible endpoint) usage: the
# tools/ tool-use loop AND the agent's DECIDE step (agent/decision.py). Pinned
# rather than commented out so that a single "pip install -r requirements.txt"
# leaves nothing to add later; the import inside the code is still lazy, so the
# offline demo and the whole test suite run whether or not the SDK is present.
anthropic==1.8.0

# Twilio and the Meta WhatsApp Cloud API are called over plain HTTP by
# tools/providers.py (see tools/transport.py, which uses stdlib urllib), so
# their SDKs are never imported and never needed -- these lines stay commented
# to record that the choice was deliberate. If you add code that imports one,
# uncomment it and pin it here.
# twilio==8.11.0
# python-whatsapp==0.0.1

# AWS messaging backends for the send_sms / send_email tools (AWS End User
# Messaging SMS + Amazon SES). Imported lazily: without it those two tools
# report error_code "config_missing" and everything else keeps working.
boto3==1.35.36
# Imported directly (botocore.exceptions.ClientError) by scripts/ses_domain_setup.py
# and the AWS tool tests, so it is pinned here rather than left to arrive
# transitively through boto3. 1.35.99 is the version boto3 1.35.36 resolves and
# satisfies its "<1.36.0,>=1.35.36" requirement.
botocore==1.35.99

# Reads .xlsx patient lists uploaded to the dashboard. Imported lazily inside
# utils/llm_parser._parse_excel: without it an .xlsx upload returns a clear
# "pip install openpyxl" message and CSV/JSON/TXT uploads keep working.
openpyxl==3.1.2

# Development and Testing
pytest==7.4.3
pytest-cov==4.1.0

# TLS trust store for outbound HTTPS/SMTP. Optional at import time (tools/tls.py
# falls back to the system store), but a macOS python.org install ships no CAs
# until "Install Certificates" is run, so every live send fails with
# CERTIFICATE_VERIFY_FAILED without it.
certifi==2026.7.22

# Code Quality
flake8==6.1.0
black==23.12.1

# Type Checking
mypy==1.7.1

# Documentation
sphinx==7.2.6
